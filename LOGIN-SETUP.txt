ตั้งค่า Login (ต้องทำใน Supabase ก่อนเริ่มใช้งาน)

1. เปิดโปรเจกต์ jqkhnnmcfzmmpursivxh > SQL Editor
   รัน supabase-schema.sql ฉบับนี้ทั้งไฟล์ เพื่อปิดสิทธิ์แบบไม่ Login
   ข้อมูลเดิมไม่ถูกลบ
2. ไป Authentication > Users > Add user สร้างบัญชีอีเมล/รหัสผ่าน
   เปิด Auto Confirm User สำหรับบัญชีที่ผู้ดูแลสร้าง
3. กำหนดสิทธิ์ใน SQL Editor ด้วยตัวอย่างด้านล่าง เปลี่ยนอีเมลเป็นบัญชีจริง

-- ส่วนกลาง: จัดการสูตร นำเข้า ล้างข้อมูล และดูทุกสาขา
insert into public.bom_members(user_id,role,branch)
select id,'admin',null from auth.users where email='ADMIN_EMAIL_HERE'
on conflict(user_id) do update set role=excluded.role,branch=excluded.branch;

-- สาขา: คีย์ยอดนับและดูรายงานของสาขาตนเอง
-- เปลี่ยน 38 เป็นรหัสสาขาที่ต้องการ ไม่ใช่ชื่อสาขา
insert into public.bom_members(user_id,role,branch)
select id,'branch','38' from auth.users where email='BRANCH_EMAIL_HERE'
on conflict(user_id) do update set role=excluded.role,branch=excluded.branch;

ตรวจว่ามีแถวสมาชิกจริง ไม่ใช่ผลลัพธ์ 0 rows เพราะอีเมลไม่ตรง:
select m.*,u.email from public.bom_members m join auth.users u on u.id=m.user_id;

4. รีเฟรชหน้าเว็บ แล้วเข้าสู่ระบบด้วยบัญชีที่สร้าง
   โทเคนเก็บในหน่วยความจำเท่านั้น รีเฟรชหน้าต้องเข้าสู่ระบบใหม่
   หากลืมรหัสผ่าน ให้ผู้ดูแลจัดการบัญชีผ่าน Supabase Authentication

สำคัญ: เปลี่ยนหน้าเว็บอย่างเดียวไม่ปิด API เก่า ต้องรัน SQL ข้อ 1 ด้วย
ไม่มี Secret Key หรือ Service Role Key ในหน้าเว็บ
บัญชีที่ไม่มีแถวใน bom_members จะเข้าใช้งานไม่ได้
คู่มือ Supabase: https://supabase.com/docs/guides/auth/passwords
